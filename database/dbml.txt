Table employees {
  id integer [primary key]
  employee_code varchar [unique]
  first_name varchar
  last_name varchar
  date_of_birth date
  gender varchar
  email varchar [unique]
  phone varchar
  address text
  hire_date date
  status varchar
  department_id integer [not null]
  position_id integer [not null]
  created_at timestamp
  updated_at timestamp
}

Table departments {
  id integer [primary key]
  name varchar
  code varchar [unique]
  description text
  created_at timestamp
}

Table positions {
  id integer [primary key]
  name varchar
  code varchar [unique]
  level varchar
  description text
  created_at timestamp
}

Table contracts {
  id integer [primary key]
  employee_id integer [not null]
  contract_type varchar
  start_date date
  end_date date
  salary decimal
  status varchar
  document_url varchar
  created_at timestamp
}

Table employee_events {
  id integer [primary key]
  employee_id integer [not null]
  event_type varchar
  effective_date date
  old_department_id integer
  new_department_id integer
  old_position_id integer
  new_position_id integer
  description text
  created_by integer
  created_at timestamp
}

Table onboarding_tasks {
  id integer [primary key]
  employee_id integer [not null]
  task_name varchar
  description text
  assigned_to integer
  due_date date
  status varchar
  completed_at timestamp
  created_at timestamp
}

Table employee_documents {
  id integer [primary key]
  employee_id integer [not null]
  document_type varchar
  file_name varchar
  file_url varchar
  uploaded_by integer
  uploaded_at timestamp
}

Table job_postings {
  id integer [primary key]
  title varchar
  department_id integer [not null]
  description text
  requirements text
  location varchar
  employment_type varchar
  salary_min decimal
  salary_max decimal
  status varchar
  published_at timestamp
  closing_date date
  created_by integer
  created_at timestamp
  updated_at timestamp
}

Table candidates {
  id integer [primary key]
  first_name varchar
  last_name varchar
  email varchar [unique]
  phone varchar
  address text
  linkedin_url varchar
  portfolio_url varchar
  created_at timestamp
  updated_at timestamp
}

Table resumes {
  id integer [primary key]
  candidate_id integer [not null]
  file_name varchar
  file_url varchar
  parsed_text text
  parsed_data json
  uploaded_at timestamp
}

Table applications {
  id integer [primary key]
  candidate_id integer [not null]
  job_posting_id integer [not null]
  resume_id integer
  status varchar
  applied_at timestamp
  source varchar
  notes text
  created_at timestamp
  updated_at timestamp
}

Table interviews {
  id integer [primary key]
  application_id integer [not null]
  interview_type varchar
  scheduled_at timestamp
  location varchar
  meeting_url varchar
  interviewer_id integer
  status varchar
  notes text
  created_at timestamp
}

Table evaluations {
  id integer [primary key]
  interview_id integer [not null]
  evaluator_id integer [not null]
  technical_score integer
  communication_score integer
  problem_solving_score integer
  teamwork_score integer
  overall_score decimal
  recommendation varchar
  feedback text
  created_at timestamp
}

Table offers {
  id integer [primary key]
  application_id integer [not null]
  salary decimal
  employment_type varchar
  start_date date
  expiration_date date
  status varchar
  offer_letter_url varchar
  created_at timestamp
  updated_at timestamp
}

// ==============================
// Attendance & Leave Management
// ==============================

// Work Shifts
Table work_shifts {
  id integer [primary key]
  name varchar
  code varchar [unique]
  start_time time
  end_time time
  break_duration integer
  status varchar
  created_at timestamp
  updated_at timestamp
}

// Employee Shift Assignment
Table employee_shifts {
  id integer [primary key]
  employee_id integer [not null]
  shift_id integer [not null]
  start_date date
  end_date date
  status varchar
  created_at timestamp
  updated_at timestamp
}

// Work Schedule
Table work_schedules {
  id integer [primary key]
  employee_id integer [not null]
  shift_id integer [not null]
  work_date date
  status varchar
  created_at timestamp
}

// Attendance Records
Table attendance_records {
  id integer [primary key]
  employee_id integer [not null]
  work_date date
  check_in timestamp
  check_out timestamp
  late_minutes integer
  early_leave_minutes integer
  total_work_hours decimal
  status varchar
  notes text
  created_at timestamp
  updated_at timestamp
}

// Leave Types
Table leave_types {
  id integer [primary key]
  name varchar
  code varchar [unique]
  description text
  max_days integer
  status varchar
  created_at timestamp
}

// Leave Requests
Table leave_requests {
  id integer [primary key]
  employee_id integer [not null]
  leave_type_id integer [not null]
  start_date date
  end_date date
  total_days decimal
  reason text
  status varchar
  created_at timestamp
  updated_at timestamp
}

// Leave Approvals
Table leave_approvals {
  id integer [primary key]
  leave_request_id integer [not null]
  approver_id integer [not null]
  action varchar
  comment text
  approved_at timestamp
  created_at timestamp
}

Ref job_department: job_postings.department_id > departments.id

Ref candidate_resumes: resumes.candidate_id > candidates.id

Ref candidate_applications: applications.candidate_id > candidates.id

Ref job_applications: applications.job_posting_id > job_postings.id

Ref application_resume: applications.resume_id > resumes.id

Ref application_interviews: interviews.application_id > applications.id

Ref interview_evaluations: evaluations.interview_id > interviews.id

Ref application_offers: offers.application_id > applications.id

Ref employee_department: employees.department_id > departments.id // many-to-one

Ref employee_position: employees.position_id > positions.id // many-to-one

Ref employee_contracts: contracts.employee_id > employees.id // many-to-one

Ref employee_events: employee_events.employee_id > employees.id // many-to-one

Ref employee_onboarding: onboarding_tasks.employee_id > employees.id // many-to-one

Ref employee_documents: employee_documents.employee_id > employees.id // many-to-one

// Work Shifts
Ref employee_shift_employee: employee_shifts.employee_id > employees.id
Ref employee_shift_shift: employee_shifts.shift_id > work_shifts.id

// Work Schedule
Ref schedule_employee: work_schedules.employee_id > employees.id
Ref schedule_shift: work_schedules.shift_id > work_shifts.id

// Attendance
Ref attendance_employee: attendance_records.employee_id > employees.id

// Leave
Ref leave_type: leave_requests.leave_type_id > leave_types.id
Ref leave_employee: leave_requests.employee_id > employees.id

// Leave Approval
Ref leave_approval_request: leave_approvals.leave_request_id > leave_requests.id
Ref leave_approval_approver: leave_approvals.approver_id > employees.id
